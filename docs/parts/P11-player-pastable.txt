XE P11 — WINDOWS LIBMPV PLAYER + STREAMX TV OVERLAY
====================================================

Pasteable source bundle for the existing XE P1–P10 tree. Merge CMake changes; do not replace the root CMakeLists.txt. The backend uses libmpv's render API with OpenGL and renders into an XE-owned GL texture/FBO. Because P3's public render API does not expose an external texture/FBO import operation, this bundle includes an explicit small P3 extension contract (`xe_r_external_texture`) and an OpenGL bridge callback; wire that bridge to your renderer's texture draw path. Do not silently substitute a CPU screenshot or a second window.

Important: the code is an integration scaffold and has not been compiled against your exact P3/Windows/libmpv SDK. The OpenGL render callback must execute while the host's GL context is current. The Windows host must link mpv and OpenGL, and provide a clock callback if you want a localized clock instead of the fallback elapsed display.

#ifdef _WIN32
#define WIN32_LEAN_AND_MEAN
#include <windows.h>
#endif
#include <GL/gl.h>
#include <mpv/client.h>
#include <mpv/render.h>
#include <mpv/render_gl.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include "player.h"
#include "player_mpv.h"
#include "xe/render.h"

/* Renderer bridge: implement this in P3's GL renderer so an existing GL texture
   can be sampled with the normal XE rounded-image pipeline. Return 0 on failure. */
#if defined(_WIN32)
__declspec(dllimport) int xe_r_external_texture(const XeTexture *texture, unsigned int gl_name, int width, int height);
#else
extern int xe_r_external_texture(const XeTexture *texture, unsigned int gl_name, int width, int height);
#endif

typedef struct {
    mpv_handle *mpv;
    mpv_render_context *render;
    GLuint fbo, texture;
    int width, height, initialized, needs_redraw;
    XeTexture *xe_texture;
} MpvState;
static MpvState M;

static void mpv_wakeup(void *ctx) { (void)ctx; M.needs_redraw=1; }
static void *mpv_get_proc(void *ctx, const char *name) {
    (void)ctx;
#ifdef _WIN32
    void *p=(void *)wglGetProcAddress(name);
    if(!p) { static HMODULE gl; if(!gl)gl=LoadLibraryA("opengl32.dll"); if(gl)p=(void *)GetProcAddress(gl,name); }
    return p;
#else
    (void)name; return NULL;
#endif
}
static void destroy_target(void) {
    if(M.xe_texture) { /* P3 owns the wrapper; it must not delete the GL name twice. */ M.xe_texture=NULL; }
    if(M.fbo)glDeleteFramebuffers(1,&M.fbo);
    if(M.texture)glDeleteTextures(1,&M.texture);
    M.fbo=M.texture=0;M.width=M.height=0;
}
static int ensure_target(int w,int h) {
    if(w<1||h<1)return 0;
    if(M.width==w&&M.height==h&&M.fbo&&M.texture)return 1;
    destroy_target();
    glGenTextures(1,&M.texture);glBindTexture(GL_TEXTURE_2D,M.texture);
    glTexParameteri(GL_TEXTURE_2D,GL_TEXTURE_MIN_FILTER,GL_LINEAR);
    glTexParameteri(GL_TEXTURE_2D,GL_TEXTURE_MAG_FILTER,GL_LINEAR);
    glTexParameteri(GL_TEXTURE_2D,GL_TEXTURE_WRAP_S,GL_CLAMP_TO_EDGE);
    glTexParameteri(GL_TEXTURE_2D,GL_TEXTURE_WRAP_T,GL_CLAMP_TO_EDGE);
    glTexImage2D(GL_TEXTURE_2D,0,GL_RGBA,w,h,0,GL_RGBA,GL_UNSIGNED_BYTE,NULL);
    glGenFramebuffers(1,&M.fbo);glBindFramebuffer(GL_FRAMEBUFFER,M.fbo);
    glFramebufferTexture2D(GL_FRAMEBUFFER,GL_COLOR_ATTACHMENT0,GL_TEXTURE_2D,M.texture,0);
    if(glCheckFramebufferStatus(GL_FRAMEBUFFER)!=GL_FRAMEBUFFER_COMPLETE){glBindFramebuffer(GL_FRAMEBUFFER,0);destroy_target();return 0;}
    glBindFramebuffer(GL_FRAMEBUFFER,0);M.width=w;M.height=h;
    /* The wrapper is a non-owning view of the GL texture. P3 bridge must honor this. */
    M.xe_texture=(XeTexture *)1;
    return 1;
}
int xe_player_mpv_init(void) {
    int yes=1;mpv_opengl_init_params gl_init={mpv_get_proc,NULL};
    mpv_render_param params[]={ {MPV_RENDER_PARAM_API_TYPE,(void *)MPV_RENDER_API_TYPE_OPENGL},
      {MPV_RENDER_PARAM_OPENGL_INIT_PARAMS,&gl_init},{MPV_RENDER_PARAM_INVALID,NULL} };
    memset(&M,0,sizeof(M));M.mpv=mpv_create();if(!M.mpv)return 0;
    mpv_set_option_string(M.mpv,"vo","libmpv");
    mpv_set_option_string(M.mpv,"hwdec","auto-safe");
    mpv_set_option_string(M.mpv,"cache","yes");
    mpv_set_option_string(M.mpv,"demuxer-max-bytes","150MiB");
    mpv_set_option_string(M.mpv,"demuxer-max-back-bytes","50MiB");
    mpv_set_option(M.mpv,"keep-open",MPV_FORMAT_FLAG,&yes);
    if(mpv_initialize(M.mpv)<0){mpv_terminate_destroy(M.mpv);memset(&M,0,sizeof(M));return 0;}
    if(mpv_render_context_create(&M.render,M.mpv,params)<0){mpv_terminate_destroy(M.mpv);memset(&M,0,sizeof(M));return 0;}
    mpv_render_context_set_update_callback(M.render,mpv_wakeup,NULL);M.initialized=1;return 1;
}
void xe_player_mpv_render(int width,int height) {
    if(!M.initialized||!M.render||!ensure_target(width,height))return;
    uint32_t fbo=(uint32_t)M.fbo;
    mpv_opengl_fbo mpvfbo={ (int)fbo,width,height,0 };
    int flip=1;
    mpv_render_param rp[]={ {MPV_RENDER_PARAM_OPENGL_FBO,&mpvfbo},
      {MPV_RENDER_PARAM_FLIP_Y,&flip},{MPV_RENDER_PARAM_INVALID,NULL} };
    glBindFramebuffer(GL_FRAMEBUFFER,M.fbo);glViewport(0,0,width,height);
    mpv_render_context_render(M.render,rp);
    glBindFramebuffer(GL_FRAMEBUFFER,0);
    /* Renderer integration must draw M.texture. See README P11. */
    xe_r_external_texture(M.xe_texture,M.texture,width,height);
    M.needs_redraw=0;
}
void xe_player_mpv_shutdown(void) {
    if(M.render){mpv_render_context_set_update_callback(M.render,NULL,NULL);mpv_render_context_free(M.render);M.render=NULL;}
    if(M.mpv){mpv_terminate_destroy(M.mpv);M.mpv=NULL;}destroy_target();memset(&M,0,sizeof(M));
}
#endif

### FILE: README_P11.md
```text
P11 — Windows libmpv render API integration

1. Install a libmpv Windows SDK matching the app architecture and compiler. Ensure mpv/client.h, mpv/render.h, mpv/render_gl.h, the import library, and runtime DLL are available.
2. Add source files and merge CMakeLists_P11_integration.txt.
3. Extend P3 with `xe_r_external_texture(...)` so it draws the provided GL texture name through the existing shader/batching pipeline. The XE texture wrapper must be non-owning; player_mpv.c deletes its GL texture/FBO itself.
4. Fix the ownership seam: player commands and render context must use the exact same mpv_handle. The draft highlights this but deliberately does not pretend the two-handle version is correct.
5. In the app frame, keep the GL context current; render the mpv frame into its FBO, draw that texture, then draw the overlay. Do not call xe_r_begin/end twice in the same frame; integrate xe_player_frame as an overlay-only call inside the host's one render pass.
6. Do not call xe_platform_poll from two places. In production, the host owns event polling and passes events into a shared UI pump; overlay should not independently consume events if the app also polls them.
7. Test with the public HLS URL in tests/p11_player_test.c. Confirm video/audio, start position, pause/resume, relative seeking, track list, and buffer state.

Current limitations / explicit integration work:
- This draft is not compile-verified against the exact libmpv SDK or P3 renderer.
- `xe_r_external_texture` is a required P3 bridge and is not in the P3 public header yet.
- Consolidate mpv_handle ownership before use; the sample currently documents the seam but does not fully consolidate it.
- `http-header-fields` is generated from name/value pairs; escaping commas/newlines needs hardening for arbitrary untrusted values.
- Overlay clock is a placeholder label; Episodes/Next episode/More open menu state only and need app callbacks.
- The buffering card draws a moving border and heartbeat-style line; percentage is not available from the public mpv property API used here and is not displayed as a fabricated value.
- Android backend is intentionally out of scope for P11; it can implement the same `player.h` contract with ExoPlayer later.
```
