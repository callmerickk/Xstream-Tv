cmake_minimum_required(VERSION 3.16)
project(xstream_tv LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

option(XE_GLES "Use OpenGL ES 2 (Android)" OFF)
option(XE_HEADLESS "Build the library without OpenGL (compile checks only)" OFF)
option(XE_BUILD_DESKTOP "Build the SDL2 desktop app (Windows, Linux)" ON)
option(XE_WITH_WEBP "Decode WebP images with libwebp" OFF)
if(ANDROID)
  set(XE_GLES ON CACHE BOOL "" FORCE)
  set(XE_BUILD_DESKTOP OFF CACHE BOOL "" FORCE)
endif()

# ---------------------------------------------------------------- engine library
file(GLOB_RECURSE XE_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.c)
if(ANDROID)
  # Android has its own network (OkHttp) and player (ExoPlayer) backends under android/.
  list(FILTER XE_SOURCES EXCLUDE REGEX "src/net/net\\.c$|src/player/player\\.c$")
endif()
add_library(xe STATIC ${XE_SOURCES} third_party/cjson/cJSON.c)
target_include_directories(xe PUBLIC include
  PRIVATE third_party/cjson third_party/stb third_party/mpv/include)
set_target_properties(xe PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(XE_GLES)
  target_compile_definitions(xe PUBLIC XE_GLES=1)
endif()
if(XE_HEADLESS)
  target_compile_definitions(xe PRIVATE XE_RENDER_NO_GL=1)
endif()
if(MSVC)
  target_compile_options(xe PRIVATE /W3 /utf-8)
  target_compile_definitions(xe PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(xe PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-sign-compare -Wno-missing-field-initializers)
endif()

find_package(Threads REQUIRED)
if(ANDROID)
  # FreeType and HarfBuzz are built from source for each Android ABI.
  enable_language(CXX)
  include(FetchContent)
  set(FT_DISABLE_ZLIB ON CACHE BOOL "" FORCE)
  set(FT_DISABLE_BZIP2 ON CACHE BOOL "" FORCE)
  set(FT_DISABLE_PNG ON CACHE BOOL "" FORCE)
  set(FT_DISABLE_HARFBUZZ ON CACHE BOOL "" FORCE)
  set(FT_DISABLE_BROTLI ON CACHE BOOL "" FORCE)
  FetchContent_Declare(freetype GIT_REPOSITORY https://github.com/freetype/freetype.git GIT_TAG VER-2-13-3 GIT_SHALLOW TRUE)
  FetchContent_Declare(harfbuzz GIT_REPOSITORY https://github.com/harfbuzz/harfbuzz.git GIT_TAG 10.1.0 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(freetype)
  FetchContent_GetProperties(harfbuzz)
  if(NOT harfbuzz_POPULATED)
    FetchContent_Populate(harfbuzz)
  endif()
  add_library(xe_harfbuzz STATIC ${harfbuzz_SOURCE_DIR}/src/harfbuzz.cc)
  target_include_directories(xe_harfbuzz PUBLIC ${harfbuzz_SOURCE_DIR}/src)
  target_compile_definitions(xe_harfbuzz PRIVATE HAVE_FREETYPE=1)
  target_compile_options(xe_harfbuzz PRIVATE -fno-exceptions -fno-rtti -Os)
  target_link_libraries(xe_harfbuzz PUBLIC freetype)
  target_link_libraries(xe PUBLIC Threads::Threads freetype xe_harfbuzz)
else()
  find_package(Freetype REQUIRED)
  find_package(CURL REQUIRED)
  find_package(harfbuzz CONFIG QUIET)
  if(TARGET harfbuzz::harfbuzz)
    set(XE_HB harfbuzz::harfbuzz)
  else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(HB REQUIRED IMPORTED_TARGET harfbuzz)
    set(XE_HB PkgConfig::HB)
  endif()
  target_link_libraries(xe PUBLIC Threads::Threads Freetype::Freetype ${XE_HB} CURL::libcurl)
endif()
if(XE_WITH_WEBP)
  find_package(WebP CONFIG REQUIRED)
  target_link_libraries(xe PUBLIC WebP::webp WebP::webpdecoder)
  target_compile_definitions(xe PRIVATE XE_HAVE_WEBP=1)
endif()
if(UNIX)
  target_link_libraries(xe PUBLIC m ${CMAKE_DL_LIBS})
endif()
if(ANDROID)
  target_link_libraries(xe PUBLIC log GLESv2)
elseif(NOT XE_HEADLESS)
  find_package(OpenGL REQUIRED)
  target_link_libraries(xe PUBLIC OpenGL::GL)
endif()

# ---------------------------------------------------------------- desktop app
if(XE_BUILD_DESKTOP AND NOT XE_HEADLESS)
  find_package(SDL2 CONFIG REQUIRED)
  if(WIN32)
    # The Xstream mark as the .exe icon (Explorer, taskbar, Start menu).
    enable_language(RC)
    add_executable(xstream hosts/desktop/main_sdl.c windows/xstream.rc)
  else()
    add_executable(xstream hosts/desktop/main_sdl.c)
  endif()
  target_include_directories(xstream PRIVATE third_party/stb)
  if(TARGET SDL2::SDL2)
    target_link_libraries(xstream PRIVATE xe SDL2::SDL2)
  else()
    target_link_libraries(xstream PRIVATE xe SDL2::SDL2-static)
  endif()
  # The app's lifecycle functions live in the library; pull them in.
  if(MSVC)
    target_link_options(xstream PRIVATE /INCLUDE:xe_app_init /SUBSYSTEM:WINDOWS /ENTRY:mainCRTStartup)
    target_compile_options(xstream PRIVATE /utf-8)
  else()
    target_link_options(xstream PRIVATE -Wl,--undefined=xe_app_init)
  endif()
  add_custom_command(TARGET xstream POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/assets $<TARGET_FILE_DIR:xstream>/assets)
  if(WIN32 AND CMAKE_VERSION VERSION_GREATER_EQUAL 3.21)
    add_custom_command(TARGET xstream POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:xstream> $<TARGET_FILE_DIR:xstream>
      COMMAND_EXPAND_LISTS)
  endif()
endif()

# ---------------------------------------------------------------- Android host library
if(ANDROID)
  set(XE_AH ${CMAKE_CURRENT_SOURCE_DIR}/android/xehost/src/main/cpp)
  file(GLOB XE_ANDROID_NET ${XE_AH}/net_android*.c)
  # Prefer the real OkHttp backend once it exists; the stub only keeps the build linking.
  list(FILTER XE_ANDROID_NET EXCLUDE REGEX "net_android_stub\\.c$")
  if(NOT XE_ANDROID_NET)
    set(XE_ANDROID_NET ${XE_AH}/net_android_stub.c)
  endif()
  add_library(xe_android SHARED ${XE_AH}/xe_jni.c ${XE_AH}/player_android.c ${XE_ANDROID_NET})
  target_include_directories(xe_android PRIVATE ${XE_AH})
  target_link_libraries(xe_android PRIVATE xe android log GLESv2 EGL)
  # The engine's lifecycle functions live in libxe; keep the whole library.
  target_link_options(xe_android PRIVATE -Wl,--undefined=xe_app_init -Wl,--undefined=xe_app_frame -Wl,--undefined=xe_app_shutdown)
endif()
